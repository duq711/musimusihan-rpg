  109 | static func _fit_segment(rest_start: Vector3, rest_end: Vector3, target_start: Vector3, target_end: Vector3) -> Transform3D:
  110 | 	var rest_delta := rest_end - rest_start
  111 | 	var target_delta := target_end - target_start
  112 | 	if minf(rest_delta.length(), target_delta.length()) < 0.00001:
  113 | 		return Transform3D.IDENTITY
  114 | 	var axis := rest_delta.normalized()
  115 | 	var axial_change := target_delta.length() / rest_delta.length() - 1.0
  116 | 	var axial_scale := Basis(
  117 | 		Vector3.RIGHT + axis * axis.x * axial_change,
  118 | 		Vector3.UP + axis * axis.y * axial_change,
  119 | 		Vector3.BACK + axis * axis.z * axial_change
  120 | 	)
  121 | 	# The shortest rotation avoids the reference-axis roll changes of a look-at
  122 | 	# frame, while axial scaling keeps the two actual sleeve endpoints joined.
  123 | 	var target_axis := target_delta.normalized()
  124 | 	var cross := axis.cross(target_axis)
  125 | 	var dot := clampf(axis.dot(target_axis), -1.0, 1.0)
  126 | 	var rotation := Quaternion.IDENTITY
  127 | 	# Quaternion(from, to) treats tiny angles as identity. That leaves visible
  128 | 	# segments short of their shared endpoint while the guard starts to rise.
  129 | 	if cross.length() > 0.000000001:
  130 | 		rotation = Quaternion(cross.normalized(), atan2(cross.length(), dot))
  131 | 	elif dot < 0.0:
  132 | 		var perpendicular := axis.cross(Vector3.RIGHT if absf(axis.x) < 0.9 else Vector3.UP).normalized()
  133 | 		rotation = Quaternion(perpendicular, PI)
  134 | 	var orientation := Basis(rotation) * axial_scale
  135 | 	return Transform3D(orientation, target_start - orientation * rest_start)
